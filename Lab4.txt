import math
Спільні константи
M_DIV = 13 M_MUL = 16 A = 0.618033 WORDS = ["НЕ", "ТОЙ", "УРОЖАЙ", "ЩО", "В", "ПОЛІ", "А", "ТОЙ", "ЩО", "В", "КОМОРІ"]
LETTER_POSITIONS = { 'А':1, 'Б':2, 'В':3, 'Г':4, 'Ґ':5, 'Д':6, 'Е':7, 'Є':8, 'Ж':9, 'З':10, 'И':11, 'І':12, 'Ї':13, 'Й':14, 'К':15, 'Л':16, 'М':17, 'Н':18, 'О':19, 'П':20, 'Р':21, 'С':22, 'Т':23, 'У':24, 'Ф':25, 'Х':26, 'Ц':27, 'Ч':28, 'Ш':29, 'Щ':30, 'Ь':31, 'Ю':32, 'Я':33 }
Функція підрахунку суми букв (К)
def get_k(word): k = 0 for char in word: k += LETTER_POSITIONS.get(char, 0) return k
Хеш-функція ділення
def hash_div(word): return get_k(word) % M_DIV
Хеш-функція множення
def hash_mul(word): k = get_k(word) return math.floor(M_MUL * ((k * A) % 1.0))
Відкрите хешування (метод ланцюжків)
def build_open_table(words, m, hash_type): table = [[] for _ in range(m)] for w in words: if hash_type == 'div': addr = hash_div(w) else: addr = hash_mul(w) table[addr].append(w) return table
Закрите хешування (відкрита адресація)
def build_closed_table(words, m, hash_type): table = [None] * m for w in words: if hash_type == 'div': start = hash_div(w) else: start = hash_mul(w)
    for i in range(m):
        addr = (start + i) % m
        if table[addr] is None:
            table[addr] = w
            break
return table
Вивід результатів
print("--- 1. Відкрите хешування ---") print("Ділення:", build_open_table(WORDS, M_DIV, 'div')) print("Множення:", build_open_table(WORDS, M_MUL, 'mul'))
print("\n--- 2. Закрите хешування ---") print("Ділення:", build_closed_table(WORDS, M_DIV, 'div')) print("Множення:", build_closed_table(WORDS, M_MUL, 'mul'))
